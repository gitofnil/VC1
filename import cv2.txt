import cv2
import numpy as np

def calcula_snr_video(ruta_video, nom_escena):
    # 1. Obrim el vídeo
    cap = cv2.VideoCapture(ruta_video)
    frames = []
    
    # Llegim 200 frames i els passem a escala de grisos
    for _ in range(200):
        ret, frame = cap.read()
        if not ret:
            break
        # Pas a escala de grisos (nivells 0-255) i format float per no perdre precisió
        gris = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY).astype(np.float32)
        frames.append(gris)
        
    cap.release()
    
    if len(frames) < 200:
        print(f"Error: El vídeo {ruta_video} no té almenys 200 frames!")
        return
        
    frames = np.array(frames)
    
    # 2. Model de la imatge amb poc soroll (mitjana dels primers 100 frames)
    # Com que fem la mitjana, el rang teòric segueix sent de 0 a 255.
    imatge_amb_poc_soroll = np.mean(frames[0:100], axis=0)
    
    # 3. Cas A: Un sol frame (rang petit, 255)
    # Agafem el frame 101 (índex 101)
    imatge_a = frames[101]
    
    # 4. Cas B: Suma de 100 frames (rang gran, 25500)
    # Sumem els frames del 100 al 199
    imatge_b = np.sum(frames[100:200], axis=0)
    
    # --- CÀLCULS D'ERROR I SNR ---
    
    # Càlcul per a la imatge A (Rang petit)
    # Els rangs ja estan igualats (0-255)
    diferencia_a = imatge_a - imatge_amb_poc_soroll
    desviacio_a = np.std(diferencia_a)
    V_soroll_a = 2 * desviacio_a
    V_senyal_a = 255.0
    
    # Càlcul de SNR en dB (Decibels)
    # SNR = 20 * log10(Senyal / Soroll)
    snr_a = 20 * np.log10(V_senyal_a / V_soroll_a) if V_soroll_a > 0 else float('inf')
    
    # Càlcul per a la imatge B (Rang gran)
    # Ajustem els rangs: com que imatge_b és la suma de 100 frames, 
    # multipliquem el nostre "model amb poc soroll" per 100 per poder restar.
    model_ajustat = imatge_amb_poc_soroll * 100
    
    diferencia_b = imatge_b - model_ajustat
    desviacio_b = np.std(diferencia_b)
    V_soroll_b = 2 * desviacio_b
    V_senyal_b = 25500.0 # (255 * 100)
    
    snr_b = 20 * np.log10(V_senyal_b / V_soroll_b) if V_soroll_b > 0 else float('inf')
    
    # Mostrem resultats
    print(f"--- Resultats per: {nom_escena} ---")
    print(f"Desviació estàndard (Rang Petit): {desviacio_a:.2f}")
    print(f"Desviació estàndard (Rang Gran):  {desviacio_b:.2f}")
    print(f"  -> SNR Rang Petit: {snr_a:.2f} dB")
    print(f"  -> SNR Rang Gran:  {snr_b:.2f} dB\n")
    
    return imatge_a, imatge_b

# ----------------------------------------------------

# ----------------------------------------------------

print("Iniciant càlculs...\n")

# Processar escena BEN il·luminada
calcula_snr_video('ben_illuminada.mp4', 'Escena BEN il·luminada')
